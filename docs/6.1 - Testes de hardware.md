## Script de Validação Acústica Avançada (Ode à Alegria)

O script em Python abaixo foi executado na Raspberry Pi 3 para validar o controle dinâmico de frequência e ciclo de trabalho (*Duty Cycle*) via PWM por software na GPIO 17, reproduzindo notas musicais no buzzer:

```python
import RPi.GPIO as GPIO
import time

# Configuração do GPIO
BUZZER = 17

GPIO.setmode(GPIO.BCM)
GPIO.setup(BUZZER, GPIO.OUT)

# Frequências das notas musicais (Hz)
NOTAS = {
    "DO": 262,
    "RE": 294,
    "MI": 330,
    "FA": 349,
    "SOL": 392,
    "LA": 440,
    "SI": 494,
    "DO5": 523,
}

# Música: Ode à Alegria
musica = [
    ("MI", 0.4), ("MI", 0.4), ("FA", 0.4), ("SOL", 0.4),
    ("SOL", 0.4), ("FA", 0.4), ("MI", 0.4), ("RE", 0.4),
    ("DO", 0.4), ("DO", 0.4), ("RE", 0.4), ("MI", 0.4),
    ("MI", 0.6), ("RE", 0.2), ("RE", 0.8),
    ("MI", 0.4), ("MI", 0.4), ("FA", 0.4), ("SOL", 0.4),
    ("SOL", 0.4), ("FA", 0.4), ("MI", 0.4), ("RE", 0.4),
    ("DO", 0.4), ("DO", 0.4), ("RE", 0.4), ("MI", 0.4),
    ("RE", 0.6), ("DO", 0.8),
]

# Inicializa PWM
pwm = GPIO.PWM(BUZZER, 440)
pwm.start(0)

try:
    print("Tocando Ode à Alegria...")

    for nota, duracao in musica:
        if nota is None:
            pwm.ChangeDutyCycle(0)
            time.sleep(duracao)
            continue

        frequencia = NOTAS[nota]
        pwm.ChangeFrequency(frequencia)
        pwm.ChangeDutyCycle(50)
        print(f"Nota: {nota} | Frequência: {frequencia} Hz")
        time.sleep(duracao)

        pwm.ChangeDutyCycle(0)
        time.sleep(0.05)

    print("Música finalizada!")

except KeyboardInterrupt:
    print("Execução interrompida.")

finally:
    pwm.stop()
    GPIO.cleanup()
    print("GPIO liberado.")
